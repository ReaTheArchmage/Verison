// CUSTOM FLOW CONTROLS: ====================================================

    // - NIF:
    // nif is a custom flow controller, that abstracts a nested if statement:

        // from:    
        if : (x > 1)
            x += 5
            if : (x > 7)
                x -= 1
                if : (x != 8)
                    x = 0

        // to:
        nif : (x > 1), (x > 7), (x != 8)

            0* x += 5
            1* x -= 1
            2* x = 0

        // inplicit form:
        nif : (!condition), (condition)
            0* //logic here
            1* //logic here

        // explicit form:
        nif : A(not condition), B_in_A(condition)
            A* //logic here
            B_in_A* //logic here


    // - NFOR:
    // nfor is a custom flow controller, that abstracts a nested loop:

        (i32)var number : 0

        // from:
        for : (i32 x in 0.<5)
            number += 1

            for : (i32 y in 0.<10)   
                if : (y % 2 is 0)
                    number *= 2

                for : (i32 z in 0.<20)

                    if : (z % 4 is 0)
                        number *= 4

        // to:
        nfor : (i32 x in 0.<5), (i32 y in 0.<10), (i32 in 0.<20)

            0* number += 1

            1* if : (number % 2 is 0)
                number *= 2   

            2* if : (z % 4 is 0)
                number *= 4

        // inplicit form:
        nfor : loop1(i32 x in 0.<5), loop2in1(i32 y in 0.<10)
            loop1* // logic here
            loop2* // logic here
        
        // explicit form:
        nfor : (i32 x in 0.<5), (i32 y in 0.<10)
            0* // logic here
            1* // logic here
